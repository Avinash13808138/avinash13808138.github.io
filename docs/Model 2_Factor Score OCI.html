# 11. Prepare the variables for Model 2
# ============================================================

factor_model_data <- factor_model_data %>%
  rename(
    target_category = `Target Category`
  ) %>%
  mutate(
    # Binary dependent variable:
    # 0 = soft target
    # 1 = hard target
    target_category =
      as.integer(target_category),

    # Maoist is the ideology reference category
    ideology_factor = factor(
      ideology_category,
      levels = c(
        "Maoist",
        "Separatist/Ethnonationalist",
        "Islamist"
      )
    ),

    # Rest of India is the conflict-region reference
    conflict_region_factor = relevel(
      factor(
        conflict_region
      ),
      ref = "Rest of India"
    ),

    # Log-transform the one-year-lagged district intensity
    ln_lagged_district_terrorism_intensity =
      log1p(
        lagged_district_terrorism_intensity
      ),

    # Year fixed effects
    year_factor =
      factor(iyear),

    # Organization random intercept
    organization_factor =
      factor(gname)
  )

# ============================================================
# 14. Estimate Model 2 with factor-score OCI
# ============================================================

model2_factor_oci <- lme4::glmer(
  target_category ~
    ideology_factor * OCI_factor_z +
    conflict_region_factor +
    ln_lagged_district_terrorism_intensity +
    year_factor +
    (1 | organization_factor),

  data = factor_model_data,

  family = binomial(
    link = "logit"
  ),

  control = lme4::glmerControl(
    optimizer = "bobyqa",
    optCtrl = list(
      maxfun = 200000
    )
  )
)

# 17. Create the coefficient-results table

coefficient_matrix <- coef(
  summary(
    model2_factor_oci
  )
)

factor_results <- data.frame(
  term = rownames(
    coefficient_matrix
  ),

  estimate =
    coefficient_matrix[
      ,
      "Estimate"
    ],

  standard_error =
    coefficient_matrix[
      ,
      "Std. Error"
    ],

  z_value =
    coefficient_matrix[
      ,
      "z value"
    ],

  p_value =
    coefficient_matrix[
      ,
      "Pr(>|z|)"
    ],

  row.names = NULL
) %>%
  mutate(
    coefficient_ci_low =
      estimate -
      1.96 * standard_error,

    coefficient_ci_high =
      estimate +
      1.96 * standard_error,

    odds_ratio =
      exp(estimate),

    odds_ratio_ci_low =
      exp(
        coefficient_ci_low
      ),

    odds_ratio_ci_high =
      exp(
        coefficient_ci_high
      )
  )

# 19. Calculate random-intercept variance and ICC
# ============================================================

random_effects_table <- as.data.frame(
  VarCorr(
    model2_factor_oci
  )
)

  random_effects_table$vcov[1]

  sqrt(
    random_variance
  )

  random_variance /
  (
    random_variance +
