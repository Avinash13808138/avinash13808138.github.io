## 6. Multicollinearity diagnostics


#| label: component-vif

calculate_vif <- function(variable, data, predictors) {

  other_predictors <- setdiff(
    predictors,
    variable
  )

  auxiliary_formula <- as.formula(
    paste(
      variable,
      "~",
      paste(other_predictors, collapse = " + ")
    )
  )

  auxiliary_model <- lm(
    auxiliary_formula,
    data = data
  )

  auxiliary_r_squared <- summary(
    auxiliary_model
  )$r.squared

  vif_value <- 1 / (1 - auxiliary_r_squared)

  data.frame(
    Variable = variable,
    Auxiliary_R2 = auxiliary_r_squared,
    VIF = vif_value,
    Tolerance = 1 / vif_value
  )
}

component_vif <- bind_rows(

    calculate_vif,

    predictors = component_names

)

## 8. Publication-ready fixed-effects table


#| label: fixed-effects-table

fixed_effects <- tidy(
  model_components,
  effects = "fixed",
  conf.int = TRUE,
  conf.method = "Wald"
) |>
  mutate(
    odds_ratio = exp(estimate),
    odds_ratio_low = exp(conf.low),
    odds_ratio_high = exp(conf.high)
  ) |>
  filter(
    !grepl("^year", term)
  ) |>
  mutate(
    term = recode(
      term,

      "(Intercept)" =
        "Intercept",

      "ideologySeparatist/Ethnonationalist" =
        "Separatist/Ethnonationalist",

      "ideologyIslamist" =
        "Islamist",

      "prior_attacks" =
        "Prior attacks",

      "tactical_diversity" =
        "Tactical diversity",

      "geographic_spread" =
        "Geographic spread",

      "success_rate" =
        "Success rate",

      "conflict_regionKashmir" =
        "Kashmir",

      "conflict_regionMaoist-affected belt" =
        "Maoist-affected belt",

      "conflict_regionNortheast" =
        "Northeast",

      "lagged_district_intensity" =
        "Lagged district terrorism intensity"
    )
  ) |>
  select(
    Predictor = term,
    Coefficient = estimate,
    `Standard Error` = std.error,
    `95% CI Lower` = conf.low,
    `95% CI Upper` = conf.high,
    `Odds Ratio` = odds_ratio,
    `OR 95% CI Lower` = odds_ratio_low,
    `OR 95% CI Upper` = odds_ratio_high,
    `p-value` = p.value
  )
