model2_incidents <- gtd_data %>%

oci_group_year <- model2_incidents %>%
  distinct(
    gname,
    iyear,

  count(gname, iyear) %>%
  filter(n > 1)

  print("No duplicate organization-years were found.")

  print("Warning: Duplicate organization-years were found.")
