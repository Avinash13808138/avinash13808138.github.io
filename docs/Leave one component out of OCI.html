## 6. Function for estimating Model 2

#| label: estimation-function

estimate_loo_model <- function(
  data,
  index_variable,
  model_label
) {

  analysis_data <- data |>
    mutate(
      loo_oci = .data[[index_variable]]
    )

  cat(
    "\nEstimating:",
    model_label,
    "\n"
  )

  starting_time <- Sys.time()

  fitted_model <- glmer(
    hard_target ~
      ideology * loo_oci +
      conflict_region +
      ln_lagged_district_intensity +
      year +
      (1 | organization),

    data = analysis_data,

    family = binomial(
      link = "logit"
    ),

    # Faster approximation for the initial run
    nAGQ = 0,

    control = glmerControl(
      optimizer = "bobyqa",

      optCtrl = list(
        maxfun = 100000
      ),

      calc.derivs = FALSE
    )
  )

  ending_time <- Sys.time()

  cat(
    "Completed:",
    model_label,
    "\n"
  )

  cat(
    "Estimation time:",
    round(
      as.numeric(
        difftime(
          ending_time,
          starting_time,
          units = "mins"
        )
      ),
      2
    ),
    "minutes\n"
  )

  return(fitted_model)
}

## 8. Extract the OCI and interaction coefficients

#| label: coefficient-results

extract_key_coefficients <- function(
  fitted_model,
  model_label
) {

  tidy(
    fitted_model,
    effects = "fixed",
    conf.int = TRUE,
    conf.method = "Wald"
  ) |>
    filter(
      term %in% c(
        "loo_oci",
        "ideologySeparatist/Ethnonationalist:loo_oci",
        "ideologyIslamist:loo_oci"
      )
    ) |>
    mutate(
      Model = model_label,

      Predictor = recode(
        term,

        "loo_oci" =
          "OCI slope: Maoist",

        "ideologySeparatist/Ethnonationalist:loo_oci" =
          "Separatist × OCI",

        "ideologyIslamist:loo_oci" =
          "Islamist × OCI"
      ),

      Odds_Ratio =
        exp(estimate),

      OR_CI_Lower =
        exp(conf.low),

      OR_CI_Upper =
        exp(conf.high)
    ) |>
    select(
      Model,
      Predictor,
      Coefficient = estimate,
      `Standard Error` = std.error,
      `95% CI Lower` = conf.low,
      `95% CI Upper` = conf.high,
      `Odds Ratio` = Odds_Ratio,
      `OR 95% CI Lower` = OR_CI_Lower,
      `OR 95% CI Upper` = OR_CI_Upper,
      `p-value` = p.value
    )
}

## 9. Calculate ideology-specific OCI slopes

#| label: ideology-specific-slopes

extract_slopes <- function(
  fitted_model,
  model_label
) {

  trend_results <- emtrends(
    fitted_model,
    specs = ~ ideology,
    var = "loo_oci"
  )

  trend_table <- as.data.frame(
    summary(
      trend_results,
      infer = c(TRUE, TRUE),
      level = 0.95
    )
  )

  # Identify the confidence-interval column names
  lower_column <- intersect(
    c("asymp.LCL", "lower.CL"),
    names(trend_table)
  )[1]

  upper_column <- intersect(
    c("asymp.UCL", "upper.CL"),
    names(trend_table)
  )[1]

  if (
    is.na(lower_column) ||
    is.na(upper_column)
  ) {
    stop(
      paste(
        "Confidence-interval columns could not",
        "be identified in the emtrends output."
      )
    )
  }

  trend_table |>
    mutate(
      Model = model_label,

      CI_Lower =
        .data[[lower_column]],

      CI_Upper =
        .data[[upper_column]]
    ) |>
    transmute(
      Model,

      Ideology =
        ideology,

      `OCI Slope` =
        loo_oci.trend,

      `Standard Error` =
        SE,

      `95% CI Lower` =
        CI_Lower,

      `95% CI Upper` =
        CI_Upper,

      `p-value` =
        p.value
    )
}

## 10. Pairwise comparisons of ideology-specific slopes

#| label: pairwise-slope-comparisons

extract_slope_comparisons <- function(
  fitted_model,
  model_label
) {

  trend_results <- emtrends(
    fitted_model,
    specs = ~ ideology,
    var = "loo_oci"
  )

  comparison_results <- pairs(
    trend_results,
    adjust = "tukey"
  )

  comparison_table <- as.data.frame(
    summary(
      comparison_results,
      infer = c(TRUE, TRUE),
      level = 0.95,
      adjust = "tukey"
    )
  )

  # Identify the confidence-interval column names
  lower_column <- intersect(
    c("asymp.LCL", "lower.CL"),
    names(comparison_table)
  )[1]

  upper_column <- intersect(
    c("asymp.UCL", "upper.CL"),
    names(comparison_table)
  )[1]

  if (
    is.na(lower_column) ||
    is.na(upper_column)
  ) {
    stop(
      paste(
        "Confidence-interval columns could not",
        "be identified in the pairwise output."
      )
    )
  }

  comparison_table |>
    mutate(
      Model = model_label,

      CI_Lower =
        .data[[lower_column]],

      CI_Upper =
        .data[[upper_column]]
    ) |>
    transmute(
      Model,

      Comparison =
        contrast,

      `Difference in Slopes` =
        estimate,

      `Standard Error` =
        SE,

      `95% CI Lower` =
        CI_Lower,

      `95% CI Upper` =
        CI_Upper,

      `Adjusted p-value` =
        p.value
    )
}

## 11. Extract model diagnostics


#| label: diagnostics-function

extract_model_diagnostics <- function(
  fitted_model,
  model_label
) {

  random_variance <- as.data.frame(
    VarCorr(fitted_model)
  )$vcov[1]

  random_sd <-
    sqrt(random_variance)

  icc <- random_variance /
    (
      random_variance +
      (pi^2 / 3)
    )

  optimizer_code <-
    fitted_model@optinfo$conv$opt

  convergence_messages <-
    fitted_model@optinfo$conv$lme4$messages

  data.frame(
    Model = model_label,

    Observations =
      nobs(fitted_model),

    Organizations =
      length(
        unique(
          fitted_model@frame$organization
        )
      ),

    Log_Likelihood =
      as.numeric(
        logLik(fitted_model)
      ),

    AIC =
      AIC(fitted_model),

    BIC =
      BIC(fitted_model),

    Random_Intercept_Variance =
      random_variance,

    Random_Intercept_SD =
      random_sd,

    ICC =
      icc,

    Singular =
      isSingular(
        fitted_model,
        tol = 0.0001
      ),

    Optimizer_Code =
      paste(
        optimizer_code,
        collapse = ", "
      ),

    Convergence_Warnings =
      ifelse(
        is.null(convergence_messages),
        "None",
        paste(
          convergence_messages,
          collapse = "; "
        )
      )
  )
}
