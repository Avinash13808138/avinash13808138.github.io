## Function that constructs an OCI window


#| label: oci-construction-function

construct_oci <- function(data, window_length) {

  organization_years <- data |>
    distinct(gname, iyear) |>
    arrange(gname, iyear)

  calculate_one_year <- function(group_name, focal_year) {

    historical_data <- data |>
      filter(
        gname == group_name,
        iyear >= focal_year - window_length,
        iyear <= focal_year - 1
      )

    prior_attacks <- nrow(historical_data)

    if (prior_attacks == 0) {
      return(
        data.frame(
          gname = group_name,
          iyear = focal_year,
          prior_attacks = NA_real_,
          tactical_diversity = NA_real_,
          geographic_spread = NA_real_,
          adjusted_success_rate = NA_real_
        )
      )
    }

    # Exclude GTD attack type 9, which is Unknown
    known_tactics <- historical_data |>
      filter(
        !is.na(attacktype1),
        attacktype1 != 9
      )

    if (nrow(known_tactics) == 0) {

      tactical_diversity <- NA_real_

    } else {

      tactic_counts <- table(
        known_tactics$attacktype1
      )

      tactic_proportions <-
        as.numeric(tactic_counts) /
        sum(tactic_counts)

      shannon_entropy <- -sum(
        tactic_proportions *
          log(tactic_proportions)
      )

      # Eight known GTD attack categories
      tactical_diversity <-
        shannon_entropy / log(8)
    }

    geographic_spread <- n_distinct(
      historical_data$district_spatial,
      na.rm = TRUE
    )

    if (geographic_spread == 0) {
      geographic_spread <- NA_real_
    }

    number_successful <- sum(
      historical_data$success == 1,
      na.rm = TRUE
    )

    adjusted_success_rate <-
      (number_successful + 1) /
      (prior_attacks + 2)

    data.frame(
      gname = group_name,
      iyear = focal_year,
      prior_attacks = prior_attacks,
      tactical_diversity = tactical_diversity,
      geographic_spread = geographic_spread,
      adjusted_success_rate =
        adjusted_success_rate
    )
  }

  component_rows <- vector(
    mode = "list",
    length = nrow(organization_years)
  )

  for (i in seq_len(nrow(organization_years))) {

    component_rows[[i]] <- calculate_one_year(
      group_name =
        organization_years$gname[i],

      focal_year =
        organization_years$iyear[i]
    )
  }

  components <- bind_rows(component_rows) |>
    filter(
      if_all(
        c(
          prior_attacks,
          tactical_diversity,
          geographic_spread,
          adjusted_success_rate
        ),
        ~ !is.na(.x)
      )
    ) |>
    mutate(
      ln_prior_attacks =
        log1p(prior_attacks),

      ln_geographic_spread =
        log1p(geographic_spread),

      z_prior_attacks =
        as.numeric(scale(ln_prior_attacks)),

      z_tactical_diversity =
        as.numeric(scale(tactical_diversity)),

      z_geographic_spread =
        as.numeric(scale(ln_geographic_spread)),

      z_success_rate =
        as.numeric(scale(adjusted_success_rate)),

      OCI = rowMeans(
        cbind(
          z_prior_attacks,
          z_tactical_diversity,
          z_geographic_spread,
          z_success_rate
        )
      ),

      window = window_length
    )

  return(components)
}

## Estimate the three common-sample models


#| label: window-model-function

estimate_window_model <- function(
  data,
  oci_variable,
  model_label
) {

  model_data <- data |>
    mutate(
      OCI_window =
        .data[[oci_variable]]
    )

  fitted_model <- glmer(
    hard_target ~
      ideology * OCI_window +
      conflict_region +
      ln_lagged_district_intensity +
      year +
      (1 | organization),

    data = model_data,

    family = binomial(
      link = "logit"
    ),

    # Fast initial estimation
    nAGQ = 0,

    control = glmerControl(
      optimizer = "bobyqa",

      optCtrl = list(
        maxfun = 100000
      ),

      calc.derivs = FALSE
    )
  )

  return(fitted_model)
}

## Extract OCI coefficients and interactions


#| label: window-coefficients

extract_window_coefficients <- function(
  model,
  model_label
) {

  tidy(
    model,
    effects = "fixed",
    conf.int = TRUE,
    conf.method = "Wald"
  ) |>
    filter(
      term %in% c(
        "OCI_window",
        "ideologySeparatist/Ethnonationalist:OCI_window",
        "ideologyIslamist:OCI_window"
      )
    ) |>
    mutate(
      Model = model_label,

      Predictor = recode(
        term,

        "OCI_window" =
          "Maoist OCI slope",

        "ideologySeparatist/Ethnonationalist:OCI_window" =
          "Separatist × OCI",

        "ideologyIslamist:OCI_window" =
          "Islamist × OCI"
      )
    ) |>
    select(
      Model,
      Predictor,
      Coefficient = estimate,
      `Standard Error` = std.error,
      `95% CI Lower` = conf.low,
      `95% CI Upper` = conf.high,
      `p-value` = p.value
    )
}

      extract_window_coefficients(

      )

  )

## Calculate ideology-specific slopes


#| label: window-slopes

extract_window_slopes <- function(
  model,
  model_label
) {

  slope_table <- as.data.frame(
    summary(
      emtrends(
        model,
        specs = ~ ideology,
        var = "OCI_window"
      ),
      infer = c(TRUE, TRUE)
    )
  )

  lower_column <- intersect(
    c("asymp.LCL", "lower.CL"),
    names(slope_table)
  )[1]

  upper_column <- intersect(
    c("asymp.UCL", "upper.CL"),
    names(slope_table)
  )[1]

  slope_table |>
    mutate(
      Model = model_label,
      CI_Lower = .data[[lower_column]],
      CI_Upper = .data[[upper_column]]
    ) |>
    transmute(
      Model,
      Ideology = ideology,
      `OCI Slope` = OCI_window.trend,
      `Standard Error` = SE,
      `95% CI Lower` = CI_Lower,
      `95% CI Upper` = CI_Upper,
      `p-value` = p.value
    )
}

## Pairwise comparisons of slopes


#| label: window-pairwise-comparisons

extract_window_comparisons <- function(
  model,
  model_label
) {

  trends <- emtrends(
    model,
    specs = ~ ideology,
    var = "OCI_window"
  )

  comparison_table <- as.data.frame(
    summary(
      pairs(
        trends,
        adjust = "tukey"
      ),
      infer = c(TRUE, TRUE)
    )
  )

  lower_column <- intersect(
    c("asymp.LCL", "lower.CL"),
    names(comparison_table)
  )[1]

  upper_column <- intersect(
    c("asymp.UCL", "upper.CL"),
    names(comparison_table)
  )[1]

  comparison_table |>
    mutate(
      Model = model_label,
      CI_Lower = .data[[lower_column]],
      CI_Upper = .data[[upper_column]]
    ) |>
    transmute(
      Model,
      Comparison = contrast,
      `Difference in Slopes` = estimate,
      `Standard Error` = SE,
      `95% CI Lower` = CI_Lower,
      `95% CI Upper` = CI_Upper,
      `Adjusted p-value` = p.value
    )
}

window_pairwise_results <- bind_rows(

      extract_window_comparisons(

      )

  )
)
