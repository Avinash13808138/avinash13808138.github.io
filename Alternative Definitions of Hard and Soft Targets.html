## Prepare each analytical sample

#| label: prepare-model-data

prepare_target_sample <- function(
  data,
  outcome_variable
) {

  data |>
    filter(
      ideology_category %in% c(
        "Maoist",
        "Separatist/Ethnonationalist",
        "Islamist"
      ),
      !is.na(OCI)
    ) |>
    transmute(
      hard_target =
        as.numeric(
          .data[[outcome_variable]]
        ),

      organization =
        factor(gname),

      organization_year =
        interaction(
          gname,
          iyear,
          drop = TRUE
        ),

      year =
        factor(iyear),

      ideology = factor(
        ideology_category,
        levels = c(
          "Maoist",
          "Separatist/Ethnonationalist",
          "Islamist"
        )
      ),

      conflict_region = factor(
        conflict_region,
        levels = c(
          "Rest of India",
          "Kashmir",
          "Maoist-affected belt",
          "Northeast"
        )
      ),

      OCI =
        as.numeric(OCI),

      ln_lagged_district_intensity =
        log1p(
          as.numeric(
            lagged_district_terrorism_intensity
          )
        )
    ) |>
    filter(
      if_all(
        everything(),
        ~ !is.na(.x)
      )
    )
}

narrow_sample <- prepare_target_sample(

summarize_target_sample <- function(
  data,
  label
) {

  data.frame(
    Definition = label,

    Incidents =
      nrow(data),

    Organizations =
      n_distinct(data$organization),

    Organization_Years =
      n_distinct(data$organization_year),

    Hard_Targets =
      sum(data$hard_target == 1),

    Soft_Targets =
      sum(data$hard_target == 0),

    Percent_Hard =
      100 * mean(data$hard_target == 1)
  )
}

## Estimate Model 2

#| label: target-model-function

estimate_target_model <- function(data) {

  glmer(
    hard_target ~
      ideology * OCI +
      conflict_region +
      ln_lagged_district_intensity +
      year +
      (1 | organization),

    data = data,

    family = binomial(
      link = "logit"
    ),

    # Faster initial analysis
    nAGQ = 0,

    control = glmerControl(
      optimizer = "bobyqa",

      optCtrl = list(
        maxfun = 100000
      ),

      calc.derivs = FALSE
    )
  )
}

## Extract OCI coefficients and interactions


#| label: target-coefficients

extract_target_coefficients <- function(
  model,
  model_label
) {

  tidy(
    model,
    effects = "fixed",
    conf.int = TRUE,
    conf.method = "Wald"
  ) |>
    filter(
      term %in% c(
        "OCI",
        "ideologySeparatist/Ethnonationalist:OCI",
        "ideologyIslamist:OCI"
      )
    ) |>
    mutate(
      Model = model_label,

      Predictor = recode(
        term,

        "OCI" =
          "Maoist OCI slope",

        "ideologySeparatist/Ethnonationalist:OCI" =
          "Separatist × OCI",

        "ideologyIslamist:OCI" =
          "Islamist × OCI"
      )
    ) |>
    select(
      Model,
      Predictor,
      Coefficient = estimate,
      `Standard Error` = std.error,
      `95% CI Lower` = conf.low,
      `95% CI Upper` = conf.high,
      `p-value` = p.value
    )
}

  )

## Ideology-specific OCI slopes


#| label: target-slopes

extract_target_slopes <- function(
  model,
  model_label
) {

  slope_table <- as.data.frame(
    summary(
      emtrends(
        model,
        specs = ~ ideology,
        var = "OCI"
      ),
      infer = c(TRUE, TRUE)
    )
  )

  lower_column <- intersect(
    c("asymp.LCL", "lower.CL"),
    names(slope_table)
  )[1]

  upper_column <- intersect(
    c("asymp.UCL", "upper.CL"),
    names(slope_table)
  )[1]

  slope_table |>
    mutate(
      Model = model_label,
      CI_Lower = .data[[lower_column]],
      CI_Upper = .data[[upper_column]]
    ) |>
    transmute(
      Model,
      Ideology = ideology,
      `OCI Slope` = OCI.trend,
      `Standard Error` = SE,
      `95% CI Lower` = CI_Lower,
      `95% CI Upper` = CI_Upper,
      `p-value` = p.value
    )
}

## Average predicted probabilities

## This uses empirical averaging over the observed covariate distributions.


#| label: average-predictions

calculate_average_predictions <- function(
  model,
  data,
  model_label
) {

  oci_grid <- seq(
    quantile(
      data$OCI,
      0.025,
      na.rm = TRUE
    ),

    quantile(
      data$OCI,
      0.975,
      na.rm = TRUE
    ),

    length.out = 15
  )

  prediction_results <- avg_predictions(
    model,

    newdata = datagrid(
      model = model,
      grid_type = "counterfactual",
      OCI = oci_grid,
      ideology = levels(data$ideology)
    ),

    by = c(
      "ideology",
      "OCI"
    ),

    re.form = NA
  )

  as.data.frame(prediction_results) |>
    mutate(
      Model = model_label
    )
}

  )
)
